#ifndef CAN_PACKET_H
#define CAN_PACKET_H

struct idPeader {
  uint8_t unused : (5+1) // 1 bit left that can be used
  uint8_t origin : 3
  uint8_t dest : 3
  uint8_t commandType : 4
} // 16 bits only 11 used

struct dataPacket {
  uint16_t unused : 8
  uint8_t hours: 8
  uint8_t minutes: 8
  uint8_t seconds: 8
  uint32_t data : 32
}

struct universalPacket {
	idPeader header;
	dataPacket data;
}

#endif \\ CAN_PACKET_H
